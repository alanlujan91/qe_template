%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Fix theorem numbering: MyST generates    %%
%% section-based counters (Theorem 1.1) but %%
%% QE requires global counters (Theorem 1). %%
%% \counterwithout is a LaTeX2e kernel      %%
%% command (2023-06-01 release or newer),   %%
%% not from etoolbox.                        %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\makeatletter
\@ifundefined{c@theorem}{}{%
  \counterwithout{theorem}{section}}
\@ifundefined{c@definition}{}{%
  \counterwithout{definition}{section}}
\@ifundefined{c@example}{}{%
  \counterwithout{example}{section}}
\@ifundefined{c@remark}{}{%
  \counterwithout{remark}{section}}
\@ifundefined{c@axiom}{}{%
  \counterwithout{axiom}{section}}
\@ifundefined{c@proposition}{}{%
  \counterwithout{proposition}{section}}
\@ifundefined{c@corollary}{}{%
  \counterwithout{corollary}{section}}
\@ifundefined{c@conjecture}{}{%
  \counterwithout{conjecture}{section}}
\@ifundefined{c@observation}{}{%
  \counterwithout{observation}{section}}
\makeatother
